/* --- 3) POPULATE DATE_DIM (pure SQL; safe wide window) --- */
USE SCHEMA CHINOOK_DB.DW;

INSERT INTO CHINOOK_DB.DW.DATE_DIM (
  DATE_KEY, FULL_DATE, DAY_NUM, WEEKDAY_ABBR, WEEKDAY_NUM,
  DAY_OF_YEAR_NUM, WEEK_OF_YEAR, MONTH_NUM, MONTH_ABBR,
  QUARTER_NUM, QUARTER_NAME, YEAR_NUM, FIRST_DAY_OF_MONTH,
  LAST_DAY_OF_MONTH, IS_WEEKEND
)
WITH seq AS (
  /* 1990-01-01 .. 2035-12-31 (adjust easily by ROWCOUNT) */
  SELECT DATEADD(day, ROW_NUMBER() OVER (ORDER BY SEQ4()) - 1, '1990-01-01'::DATE) AS d
  FROM TABLE(GENERATOR(ROWCOUNT => 17000))
)
SELECT
  TO_NUMBER(TO_CHAR(d, 'YYYYMMDD'))                 AS DATE_KEY,
  d                                                AS FULL_DATE,
  EXTRACT(DAY FROM d)                               AS DAY_NUM,
  UPPER(TO_CHAR(d, 'DY'))                           AS WEEKDAY_ABBR,
  EXTRACT(DOW FROM d)                               AS WEEKDAY_NUM,      -- 0=Sun
  EXTRACT(DOY FROM d)                               AS DAY_OF_YEAR_NUM,
  TO_NUMBER(TO_CHAR(d, 'IW'))                       AS WEEK_OF_YEAR,
  EXTRACT(MONTH FROM d)                             AS MONTH_NUM,
  UPPER(TO_CHAR(d, 'MON'))                          AS MONTH_ABBR,
  EXTRACT(QUARTER FROM d)                           AS QUARTER_NUM,
  CONCAT('Q', EXTRACT(QUARTER FROM d))              AS QUARTER_NAME,
  EXTRACT(YEAR FROM d)                              AS YEAR_NUM,
  DATE_TRUNC('MONTH', d)                            AS FIRST_DAY_OF_MONTH,
  LAST_DAY(d)                                       AS LAST_DAY_OF_MONTH,
  IFF(EXTRACT(DOW FROM d) IN (0,6), 'Y', 'N')       AS IS_WEEKEND
FROM seq
QUALIFY NOT EXISTS (
  SELECT 1 FROM CHINOOK_DB.DW.DATE_DIM t
  WHERE t.DATE_KEY = TO_NUMBER(TO_CHAR(d,'YYYYMMDD'))
);

/* --- 4) POPULATE TIME_DIM (HH:MM) --- */
INSERT INTO CHINOOK_DB.DW.TIME_DIM (TIME_KEY, HOUR_NUM, MINUTE_NUM, TIME_24_HR)
WITH h AS (SELECT SEQ4() AS h FROM TABLE(GENERATOR(ROWCOUNT => 24))),
     m AS (SELECT SEQ4() AS m FROM TABLE(GENERATOR(ROWCOUNT => 60)))
SELECT
  (h.h*100 + m.m)                                         AS TIME_KEY,
  h.h                                                     AS HOUR_NUM,
  m.m                                                     AS MINUTE_NUM,
  LPAD(h.h::STRING,2,'0') || ':' || LPAD(m.m::STRING,2,'0') AS TIME_24_HR
FROM h CROSS JOIN m
QUALIFY NOT EXISTS (
  SELECT 1 FROM CHINOOK_DB.DW.TIME_DIM t
  WHERE t.TIME_KEY = (h.h*100 + m.m)
);

/* --- 5) HELPER VIEWS IN STAGE --- */
USE SCHEMA CHINOOK_DB.STAGE;

CREATE OR REPLACE VIEW CHINOOK_DB.STAGE.VW_CUSTOMER_WITH_HASH AS
SELECT
  c.CUSTOMERID,
  c.FIRSTNAME, c.LASTNAME, c.COMPANY, c.CITY, c.STATE, c.COUNTRY, c.POSTALCODE,
  c.SUPPORTREPID,
  /* canonicalized concat with NVL + TRIM then SHA-256 */
  SHA2(
    CONCAT_WS('|',
      NVL(TRIM(c.FIRSTNAME),''),
      NVL(TRIM(c.LASTNAME),''),
      NVL(TRIM(c.COMPANY),''),
      NVL(TRIM(c.CITY),''),
      NVL(TRIM(c.STATE),''),
      NVL(TRIM(c.COUNTRY),''),
      NVL(TRIM(c.POSTALCODE),''),
      NVL(TO_VARCHAR(c.SUPPORTREPID),'')
    ), 256
  ) AS CUSTOMER_HASH_VAL
FROM CHINOOK_DB.STAGE.CUSTOMER c;

CREATE OR REPLACE VIEW CHINOOK_DB.STAGE.VW_INVOICE_KEYS AS
SELECT
  i.INVOICEID,
  i.CUSTOMERID,
  TO_DATE(i.INVOICEDATE) AS INVOICEDT,
  TO_NUMBER(TO_CHAR(TO_DATE(i.INVOICEDATE),'YYYYMMDD')) AS DATE_KEY,
  i.TOTAL
FROM CHINOOK_DB.STAGE.INVOICE i;

/* --- 6) DIMENSION LOADS (MERGE) --- */
USE SCHEMA CHINOOK_DB.DW;

/* 6a) CUSTOMER_DIM with hash-based change detection (Type-1) */
MERGE INTO CHINOOK_DB.DW.CUSTOMER_DIM d
USING CHINOOK_DB.STAGE.VW_CUSTOMER_WITH_HASH s
  ON d.CUSTOMER_ID = s.CUSTOMERID
WHEN NOT MATCHED THEN
  INSERT (
    CUSTOMER_KEY, CUSTOMER_ID, FIRST_NAME, LAST_NAME, COMPANY_NAME, CITY, STATE,
    COUNTRY, ZIP_CODE, EMPLOYEE_ID, IS_ACTIVE, SOURCE_ID, DATE_TO_WAREHOUSE, CUSTOMER_HASH_VAL
  )
  VALUES (
    CHINOOK_DB.DW.CUSTOMER_DIM_SEQ.NEXTVAL, s.CUSTOMERID, s.FIRSTNAME, s.LASTNAME, s.COMPANY, s.CITY, s.STATE,
    s.COUNTRY, s.POSTALCODE, s.SUPPORTREPID, 'Y', 'SQLSERVER_CHINOOK',
    CURRENT_TIMESTAMP(), s.CUSTOMER_HASH_VAL
  )
WHEN MATCHED AND NVL(d.CUSTOMER_HASH_VAL,'') <> NVL(s.CUSTOMER_HASH_VAL,'') THEN
  UPDATE SET
    d.FIRST_NAME        = s.FIRSTNAME,
    d.LAST_NAME         = s.LASTNAME,
    d.COMPANY_NAME      = s.COMPANY,
    d.CITY              = s.CITY,
    d.STATE             = s.STATE,
    d.COUNTRY           = s.COUNTRY,
    d.ZIP_CODE          = s.POSTALCODE,
    d.EMPLOYEE_ID       = s.SUPPORTREPID,
    d.SOURCE_ID         = 'SQLSERVER_CHINOOK',
    d.DATE_TO_WAREHOUSE = CURRENT_TIMESTAMP(),
    d.CUSTOMER_HASH_VAL = s.CUSTOMER_HASH_VAL;

/* 6b) ARTIST_DIM */
MERGE INTO CHINOOK_DB.DW.ARTIST_DIM d
USING (
  SELECT a.ARTISTID, a.NAME AS ARTISTNAME
  FROM CHINOOK_DB.STAGE.ARTIST a
) s
ON d.ARTIST_ID = s.ARTISTID
WHEN NOT MATCHED THEN
  INSERT (ARTIST_KEY, ARTIST_ID, ARTIST_NAME, SOURCE_ID, DATE_TO_WAREHOUSE)
  VALUES (CHINOOK_DB.DW.ARTIST_DIM_SEQ.NEXTVAL, s.ARTISTID, s.ARTISTNAME, 'SQLSERVER_CHINOOK', CURRENT_TIMESTAMP())
WHEN MATCHED AND NVL(d.ARTIST_NAME,'') <> NVL(s.ARTISTNAME,'') THEN
  UPDATE SET
    d.ARTIST_NAME       = s.ARTISTNAME,
    d.SOURCE_ID         = 'SQLSERVER_CHINOOK',
    d.DATE_TO_WAREHOUSE = CURRENT_TIMESTAMP();

/* --- 7) FACT LOAD (uses DATE_DIM_KEY, not raw date) --- */
INSERT INTO CHINOOK_DB.DW.SALES_FACT (
  SALES_KEY, CUSTOMER_KEY, INVOICE_ID, DATE_DIM_KEY, TOTAL_SALE_AMT, SOURCE_ID, DATE_TO_WAREHOUSE
)
SELECT
  CHINOOK_DB.DW.SALES_FACT_SEQ.NEXTVAL,
  cd.CUSTOMER_KEY,
  k.INVOICEID,
  dd.DATE_KEY,
  k.TOTAL,
  k.INVOICEID AS SOURCE_ID,
  CURRENT_TIMESTAMP()
FROM CHINOOK_DB.STAGE.VW_INVOICE_KEYS k
JOIN CHINOOK_DB.DW.DATE_DIM dd
  ON dd.DATE_KEY = k.DATE_KEY
JOIN CHINOOK_DB.DW.CUSTOMER_DIM cd
  ON cd.CUSTOMER_ID = k.CUSTOMERID
WHERE NOT EXISTS (
  SELECT 1 FROM CHINOOK_DB.DW.SALES_FACT f
  WHERE f.INVOICE_ID = k.INVOICEID
);

/* --- 8) OPTIONAL: CLUSTER FACT BY DATE FOR PRUNING --- */
ALTER TABLE CHINOOK_DB.DW.SALES_FACT CLUSTER BY (DATE_DIM_KEY);

/* --- 9) QUICK CHECKS --- */
SELECT CURRENT_DATE() AS TODAY, CURRENT_TIMESTAMP() AS NOW;

SELECT 'DATE_DIM' AS T, COUNT(*) AS ROWS FROM CHINOOK_DB.DW.DATE_DIM
UNION ALL SELECT 'TIME_DIM', COUNT(*) FROM CHINOOK_DB.DW.TIME_DIM
UNION ALL SELECT 'CUSTOMER_DIM', COUNT(*) FROM CHINOOK_DB.DW.CUSTOMER_DIM
UNION ALL SELECT 'ARTIST_DIM', COUNT(*) FROM CHINOOK_DB.DW.ARTIST_DIM
UNION ALL SELECT 'SALES_FACT', COUNT(*) FROM CHINOOK_DB.DW.SALES_FACT;

SELECT f.INVOICE_ID, f.TOTAL_SALE_AMT, d.FULL_DATE, c.COUNTRY, c.CITY
FROM CHINOOK_DB.DW.SALES_FACT f
JOIN CHINOOK_DB.DW.DATE_DIM d ON d.DATE_KEY = f.DATE_DIM_KEY
JOIN CHINOOK_DB.DW.CUSTOMER_DIM c ON c.CUSTOMER_KEY = f.CUSTOMER_KEY
ORDER BY f.INVOICE_ID
LIMIT 20;
